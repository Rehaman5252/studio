    <script>
        const brandData = [
            { brand: 'Amazon', format: 'Mixed', logoUrl: 'https://upload.wikimedia.org/wikipedia/commons/a/a9/Amazon_logo.svg', description: 'A mix of questions from all formats.' },
            { brand: 'Mastercard', format: 'IPL', logoUrl: 'https://upload.wikimedia.org/wikipedia/commons/a/a4/Mastercard_2019_logo.svg', description: 'Test your knowledge on the IPL.' },
            { brand: 'Netflix', format: 'T20', logoUrl: 'https://upload.wikimedia.org/wikipedia/commons/thumb/0/0c/Netflix_2015_N_logo.svg/1200px-Netflix_2015_N_logo.svg.png', description: 'Fast-paced questions on T20 cricket.' },
            { brand: 'ICICI', format: 'ODI', logoUrl: 'https://upload.wikimedia.org/wikipedia/commons/1/12/ICICI_Bank_Logo.svg', description: 'One Day International facts.' },
            { brand: 'Gucci', format: 'WPL', logoUrl: 'https://upload.wikimedia.org/wikipedia/commons/c/c5/Gucci_logo.svg', description: 'Questions on Women\'s Premier League.' },
            { brand: 'Nike', format: 'Test', logoUrl: 'https://upload.wikimedia.org/wikipedia/commons/a/a6/Logo_NIKE.svg', description: 'Put your classic knowledge to the test.' },
        ];

        const faceClasses = ['front', 'right', 'back', 'left', 'top', 'bottom'];
        const faceRotations = [
            { x: 0, y: 0 },    // Front
            { x: 0, y: -90 },  // Right
            { x: 0, y: -180 }, // Back
            { x: 0, y: 90 },   // Left
            { x: -90, y: 0 },  // Top
            { x: 90, y: 0 }    // Bottom
        ];

        let currentFaceIndex = 0;
        const cube = document.getElementById('cube');
        const selectedBrandCard = document.getElementById('selected-brand-card');
        const startQuizButton = document.getElementById('start-quiz-button');
        
        function updateUI(index) {
            currentFaceIndex = index;
            const brand = brandData[index];
            const rotation = faceRotations[index];

            cube.style.transform = `rotateX(${rotation.x}deg) rotateY(${rotation.y}deg)`;
            
            selectedBrandCard.innerHTML = `
                <div style="display: flex; align-items: center; gap: 1rem;">
                    <img src="${brand.logoUrl}" alt="${brand.brand}" style="width: 80px; height: 56px; object-fit: contain;">
                    <div>
                        <h3 style="margin: 0; font-size: 1.125rem;">${brand.format} Quiz</h3>
                        <p style="margin: 0; font-size: 0.875rem; color: hsl(var(--muted-foreground));">${brand.description}</p>
                    </div>
                </div>
            `;
            startQuizButton.innerText = `Start ${brand.format} Quiz`;
        }

        function createCubeFaces() {
            brandData.forEach((brand, index) => {
                const face = document.createElement('div');
                face.className = `cube-face cube-face-${faceClasses[index]}`;
                face.innerHTML = `
                    <img src="${brand.logoUrl}" alt="${brand.brand}">
                    <p style="font-weight: bold; font-size: 1.5rem; margin: 0.5rem 0 0;">${brand.format}</p>
                `;
                face.addEventListener('click', () => {
                    updateUI(index);
                    // In a real app, this would navigate or start the quiz
                    alert(`Starting ${brand.format} quiz!`);
                });
                cube.appendChild(face);
            });
        }
        
        // Timer logic
        const timerEl = document.getElementById('timer');
        function updateTimer() {
            const now = new Date();
            const minutes = now.getMinutes();
            const slotLength = 10;
            const slotEndMinute = (Math.floor(minutes / slotLength) + 1) * slotLength;
            const endTime = new Date(now);
            endTime.setMinutes(slotEndMinute, 0, 0);
            
            const diff = endTime.getTime() - now.getTime();
            const minutesLeft = String(Math.max(0, Math.floor((diff / 1000 / 60) % 60))).padStart(2, '0');
            const secondsLeft = String(Math.max(0, Math.floor((diff / 1000) % 60))).padStart(2, '0');
            
            timerEl.textContent = `${minutesLeft}:${secondsLeft}`;
        }
        setInterval(updateTimer, 1000);
        updateTimer();

        // Initial setup
        createCubeFaces();
        updateUI(0);

        // Auto-rotate cube
        setInterval(() => {
            const newIndex = (currentFaceIndex + 1) % faceRotations.length;
            updateUI(newIndex);
        }, 3000);
        
        startQuizButton.addEventListener('click', () => {
            alert(`Starting ${brandData[currentFaceIndex].format} quiz!`);
        });

    </script>
