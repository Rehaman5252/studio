/*
  Modern, singleton, SSR-safe Firebase app initialization.
  - Always use exported getter functions to get Auth/Firestore instances.
  - Ensures only one instance per browser and avoids SSR pitfalls.
*/

import { initializeApp, FirebaseApp } from 'firebase/app';
import { getAuth, Auth } from 'firebase/auth';
import { getFirestore, Firestore } from 'firebase/firestore';

let app: FirebaseApp | null = null;
let auth: Auth | null = null;
let db: Firestore | null = null;

export function getFirebaseApp(): FirebaseApp {
  if (!app && typeof window !== 'undefined') {
    app = initializeApp({
      apiKey: process.env.NEXT_PUBLIC_FIREBASE_API_KEY!,
      authDomain: process.env.NEXT_PUBLIC_FIREBASE_AUTH_DOMAIN!,
      projectId: process.env.NEXT_PUBLIC_FIREBASE_PROJECT_ID!,
      storageBucket: process.env.NEXT_PUBLIC_FIREBASE_STORAGE_BUCKET!,
      messagingSenderId: process.env.NEXT_PUBLIC_FIREBASE_MESSAGING_SENDER_ID!,
      appId: process.env.NEXT_PUBLIC_FIREBASE_APP_ID!,
    });
  }
  return app!;
}

/** Get the singleton Auth instance; always use this! */
export function getFirebaseAuth(): Auth {
  if (!auth && typeof window !== 'undefined') {
    auth = getAuth(getFirebaseApp());
  }
  return auth!;
}

/** Get the singleton Firestore instance; always use this! */
export function getFirebaseFirestore(): Firestore {
  if (!db && typeof window !== 'undefined') {
    db = getFirestore(getFirebaseApp());
  }
  return db!;
}

'use client';
/*
  AuthProvider:  
  - Manages user authentication and minimal user profile state.
  - On new users: ensures a Firestore profile doc always exists.
  - Exposes currentUser/profile for use throughout the app.
*/
import React, { createContext, useContext, useEffect, useState } from 'react';
import { onAuthStateChanged, User as FirebaseUser } from 'firebase/auth';
import { doc, setDoc, getDoc } from 'firebase/firestore';
import { getFirebaseAuth, getFirebaseFirestore } from '../lib/firebaseClient';

interface UserProfile {
  uid: string;
  displayName: string;
  email: string;
  // ...add required profile fields
}

interface AuthContextType {
  user: FirebaseUser | null;
  profile: UserProfile | null;
  loading: boolean;
  error: string | null;
}

const AuthContext = createContext<AuthContextType>({
  user: null,
  profile: null,
  loading: true,
  error: null,
});

export function AuthProvider({ children }: { children: React.ReactNode }) {
  const [user, setUser] = useState<FirebaseUser | null>(null);
  const [profile, setProfile] = useState<UserProfile | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  // Listen for auth state changes
  useEffect(() => {
    const auth = getFirebaseAuth();
    const unsub = onAuthStateChanged(auth, async (firebaseUser) => {
      setUser(firebaseUser);
      if (firebaseUser) {
        try {
          const db = getFirebaseFirestore();
          const userDocRef = doc(db, 'users', firebaseUser.uid);
          const userDoc = await getDoc(userDocRef);
          if (!userDoc.exists()) {
            // Create with sensible defaults
            const newUserProfile: UserProfile = {
              uid: firebaseUser.uid,
              displayName: firebaseUser.displayName || '',
              email: firebaseUser.email || '',
            };
            await setDoc(userDocRef, newUserProfile);
            setProfile(newUserProfile);
          } else {
            setProfile(userDoc.data() as UserProfile);
          }
        } catch (err: any) {
          setError('Failed to load your profile');
        }
      } else {
        setProfile(null);
      }
      setLoading(false);
    });
    return () => unsub();
  }, []);

  return (
    <AuthContext.Provider value={{ user, profile, loading, error }}>
      {children}
    </AuthContext.Provider>
  );
}

/** App-wide hook for access to auth context/state */
export function useAuth() {
  return useContext(AuthContext);
}

'use client';
/*
  Loads the leaderboard data on-demand when mounted.
  Shows skeleton instantly, never blocks global UI.
  Handles offline/Firestore errors gracefully.
*/
import React, { useEffect, useState } from 'react';
import { getFirebaseFirestore } from '../../lib/firebaseClient';
import { collection, getDocs } from 'firebase/firestore';

export default function LeaderboardContent() {
  const [entries, setEntries] = useState<any[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    setLoading(true);
    async function load() {
      try {
        const db = getFirebaseFirestore();
        const snap = await getDocs(collection(db, 'leaderboard'));
        setEntries(snap.docs.map(doc => doc.data()));
        setError(null);
      } catch (e: any) {
        setError(e.code === 'unavailable' || e.message?.includes('offline')
          ? 'You appear to be offline.' : 'Failed to load leaderboard');
      } finally {
        setLoading(false);
      }
    }
    load();
  }, []);

  if (loading) return <div>Loading Leaderboard...</div>;
  if (error) return <div>{error}</div>;
  return (
    <div>
      {entries.map((entry, i) => (
        <div key={i}>{entry.displayName}: {entry.score}</div>
      ))}
    </div>
  );
}

'use client';
// Per-user history, loaded only when viewing this page
import React, { useEffect, useState } from 'react';
import { useAuth } from '../../context/AuthProvider';
import { getFirebaseFirestore } from '../../lib/firebaseClient';
import { collection, getDocs } from "firebase/firestore";

export default function QuizHistoryContent() {
  const { user } = useAuth();
  const [history, setHistory] = useState<any[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    if (!user) return;
    setLoading(true);
    async function load() {
      try {
        const db = getFirebaseFirestore();
        const snap = await getDocs(collection(db, `users/${user.uid}/history`));
        setHistory(snap.docs.map(doc => doc.data()));
        setError(null);
      } catch (e: any) {
        setError(e.code === 'unavailable' || e.message?.includes('offline')
          ? 'Offline; cannot load quiz history.' : 'Could not load quiz history.');
      } finally {
        setLoading(false);
      }
    }
    load();
  }, [user]);

  if (loading) return <div>Loading Quiz History...</div>;
  if (error) return <div>{error}</div>;
  return (
    <div>
      {history.map((h, i) => <div key={i}>{h.date}: {h.score}</div>)}
    </div>
  );
}

  Fix: Never creates user doc, only allows **editing** fields in profile.
  All doc creation is centralized in AuthProvider!
*/
'use client';

import React, { useState } from 'react';
import { getFirebaseFirestore } from '../../lib/firebaseClient';
import { doc, updateDoc } from 'firebase/firestore';
import { useAuth } from '../../context/AuthProvider';

export default function CompleteProfileForm() {
  const { user, profile } = useAuth();
  const [displayName, setDisplayName] = useState(profile?.displayName || '');
  const [email, setEmail] = useState(profile?.email || '');
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const [success, setSuccess] = useState(false);

  async function onSubmit(e: React.FormEvent) {
    e.preventDefault();
    setSaving(true); setError(null); setSuccess(false);
    try {
      if (!user) throw new Error('Not signed in');
      const db = getFirebaseFirestore();
      const ref = doc(db, 'users', user.uid);
      await updateDoc(ref, { displayName, email });
      setSuccess(true);
    } catch (e: any) {
      setError('Failed to update profile');
    } finally {
      setSaving(false);
    }
  }

  if (!user) return <div>Please sign in to update your profile.</div>;

  return (
    <form onSubmit={onSubmit}>
      <input value={displayName} onChange={(e) => setDisplayName(e.target.value)} />
      <input value={email} onChange={(e) => setEmail(e.target.value)} />
      <button type="submit" disabled={saving}>
        {saving ? 'Saving...' : 'Save'}
      </button>
      {error && <div>{error}</div>}
      {success && <div>Profile updated!</div>}
    </form>
  );
}

/*
  On click: Loads quiz questions quickly and batches.
  Shows skeleton/spinner only when actually needed.
*/
'use client';

import React, { useState } from 'react';
import { getFirebaseFirestore } from '../../lib/firebaseClient';
import { collection, getDocs } from 'firebase/firestore';

export default function StartQuiz({ quizId, onReady }: { quizId: string, onReady: (questions: any[]) => void }) {
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  async function handleStart() {
    setLoading(true);
    try {
      const db = getFirebaseFirestore();
      const snap = await getDocs(collection(db, `quizzes/${quizId}/questions`));
      if (snap.empty) throw new Error('No questions found!');
      onReady(snap.docs.map(doc => doc.data()));
    } catch (e: any) {
      setError(e.code === 'unavailable' ? 'You appear offline.' : 'Failed to load quiz.');
    } finally {
      setLoading(false);
    }
  }

  return (
    <div>
      <button onClick={handleStart} disabled={loading}>
        {loading ? 'Loading...' : 'Start Quiz'}
      </button>
      {error && <div>{error}</div>}
    </div>
  );
}

/*
  Receives questions; advances per-answer.
  Calculates score instantly after final question.
  Displays result within 3 seconds, reconciles with backend if needed.

  Usage: <QuizSession questions={...} onSubmit={...} />
*/
'use client';

import React, { useState } from 'react';

export default function QuizSession({ questions, onSubmit }: { questions: any[], onSubmit: (score: number) => void }) {
  const [index, setIndex] = useState(0);
  const [answers, setAnswers] = useState<any[]>([]);
  const [score, setScore] = useState<number | null>(null);

  function answerAndAdvance(a: any) {
    const newAnswers = [...answers, a];
    setAnswers(newAnswers);

    if (index === questions.length - 1) {
      // Calculate score locally.
      const correct = newAnswers.reduce(
        (acc, ans, i) => acc + (ans === questions[i].correct ? 1 : 0),
        0
      );
      setScore(correct);
      setTimeout(() => onSubmit(correct), 1000); // Always show result in 1s
    } else {
      setIndex(index + 1);
    }
  }

  if (score !== null)
    return <div>Your score: {score} / {questions.length}</div>;

  const q = questions[index];
  return (
    <div>
      <div>{q.text}</div>
      {q.choices.map((choice: string, ci: number) =>
        <button key={ci} onClick={() => answerAndAdvance(choice)}>{choice}</button>
      )}
    </div>
  );
}
